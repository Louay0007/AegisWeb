dashboard et présentons les résultats des tests de validation.

\subsection{Middleware de sécurité frontend}

\begin{lstlisting}[style=tsstyle,
  caption={apps/web/middleware.ts — Protection routes /app et CSP},
  label={lst:middleware}]
export function middleware(request: NextRequest) {
  const nonce = generateNonce();
  const response = NextResponse.next();

  // Routes dashboard protegees
  if (request.nextUrl.pathname.startsWith('/app')) {
    const session = request.cookies.get('aegisweb_session');
    if (!session) {
      return NextResponse.redirect(new URL('/login', request.url));
    }
  }

  // CSP avec nonce (pas de unsafe-eval)
  response.headers.set('Content-Security-Policy', buildCsp(nonce));
  response.headers.set('X-Frame-Options', 'DENY');
  return response;
}
\end{lstlisting}

async enqueueWorkflowStart(runId: string, payload: StartPayload) {
  await this.workflowRunsQueue.add(
    'start-workflow-run',
    payload,
    {
      jobId: `start-${runId}`,
      attempts: 3,
      backoff: { type: 'exponential', delay: 2000 },
    },
  );
}
\end{lstlisting}
